  User Story 2 — CRUD Operations
  --------------------------------

CREATE: db.users.insertOne({ name: “Arjun”, email: “arjun@gmail.com”,
joinDate: new Date() })

db.books.insertOne({ title: “New Sci-Fi Book”, genre: “Science Fiction”,
publicationYear: 2023, authorId: ObjectId(“AUTHOR_ID”), ratings: [] })

READ: db.books.find({ genre: “Science Fiction” })

UPDATE: db.books.updateOne( { title: “Foundation” }, { $set: {
publicationYear: 2022 } } )

db.books.updateOne( { title: “Harry Potter” }, { $push: { ratings: {
user: “Vijay”, score: 5, comment: “Excellent fantasy book!” } } } )

DELETE: db.users.deleteOne({ name: “Ravi” })


  User Story 3 — Querying & Filtering
  -------------------------------------

1.  Books Published After 2015: db.books.find({ publicationYear: { $gt:
    2015 } })

2.  Authors Who Wrote Fantasy Books: db.books.distinct(“authorId”, {
    genre: “Fantasy” })

3.  Users Joined Within Last 6 Months: db.users.find({ joinDate: { $gte:
    new Date(new Date().setMonth(new Date().getMonth() - 6)) } })

4.  Books With Average Rating Greater Than 4: db.books.aggregate([ {
    $addFields: { avgRating: { $avg: "$ratings.score” } } }, { $match: {
    avgRating: { $gt: 4 } } }])
